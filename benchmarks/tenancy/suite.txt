# The whole benchmark, in the order `./run.sh suite` runs it. One step
# per line: a label, the test (a script here, without .sh), then
# VAR=value settings and the test's arguments. BACKEND picks the
# platform (vm, egres, pgsolo). Each script's header says what the
# settings mean. Edit to change a run; the copy used is saved with its
# results.

drive-idle            drive_idle

# 1. Isolation: a heavy tenant beside quiet ones.
noisy-vm              noisy      BACKEND=vm
noisy-egres           noisy      BACKEND=egres

# 2. What a tenant costs, and how many fit. A micro-VM holds about
# 85 MiB settled (tanland), so 600 is near what MEM_GIB's budget
# holds; the count stops early if not. pgsolo splits its figure into
# Postgres and the VM.
footprint-vm          footprint  BACKEND=vm      1 10 100 300 600
footprint-pgsolo      footprint  BACKEND=pgsolo  RESTART=0 1 10 100
footprint-egres       footprint  BACKEND=egres   1000 10000 50000 100000

# 3. Agents arriving and working, the micro-VMs put to sleep and woken
# as memory runs out: restored from a snapshot, or booted afresh. At
# 30 arrivals a second both sides ran out of onboarding speed before
# memory or latency gave out (tanland); 10 lets those show.
agents-vm-restore     agents     BACKEND=vm      WAKE=restore ARRIVE=10
agents-vm-boot        agents     BACKEND=vm      WAKE=boot    ARRIVE=10
agents-egres          agents     BACKEND=egres   ARRIVE=10
